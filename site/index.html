const fraud = External({
  name: "fraud.score",
  input: { amount: z.number() },
  output: { score: z.number().int() },
  attempts: 3,
  backoff: "exponential",
});

const Order = Model({
  name: "Order",
  fields: {
    amount: z.number().nonnegative(),
    status: z.enum(["draft", "confirmed"]),
  },
  flow: {
    async create(flow) {
      const scored = await flow.external(fraud, {
        amount: flow.body.amount,
      });
      if (scored.signal !== Done) return scored.signal;
      if (scored.output.score > 80) return Failed;
      flow.body.status = "confirmed";
      return Done;
    },
    async list(flow) {
      flow.filter.status.eq("confirmed");
      return Done;
    },
  },
});

GraphQL is the door.

Deep relation reads without N+1. Snapshot-consistent. Mutations return signal, id, runId. Live updates are SSE. who returns client ids.

Failure unwinds.

An exhausted external dead-letters. Then every completed step that declared a compensation is undone, not only the last one. Each undo has its own outbox row.

Analyze is a map.

Models, runs, outbox, dead letters. Token gate. Not your public surface. Never put it on the internet.
